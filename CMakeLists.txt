# Target definitions the empty mbed_configure_app_target() used to apply.
# mbed_config.cmake (generated by mbed-tools) holds ALL of them in
# MBED_TARGET_DEFINITIONS (device labels) and MBED_CONFIG_DEFINITIONS
# (every MBED_CONF_* macro). The mbed-os INTERFACE propagation is broken
# in 6.16, so apply them directly to the app target.
include(${MBED_CONFIG_PATH}/mbed_config.cmake)
target_compile_definitions(${APP_TARGET} PRIVATE
    ${MBED_TARGET_DEFINITIONS}
    ${MBED_CONFIG_DEFINITIONS}
    STM32F103xB
    USE_HAL_DRIVER
    TARGET_RTOS_MBED
    MBED_MINIMAL_PRINTF
)

# CPU/architecture flags the empty mbed_configure_app_target() used to set:
# Cortex-M3 + Thumb for the STM32F103RB. Without these, CMSIS RTX can't
# detect the architecture ("Unknown Arm Architecture!").
target_compile_options(${APP_TARGET} PRIVATE
    -mcpu=cortex-m3
    -mthumb
    -ffunction-sections
    -fdata-sections
    -fno-exceptions
    -funsigned-char
    -fomit-frame-pointer
)
target_link_options(${APP_TARGET} PRIVATE
    -mcpu=cortex-m3
    -mthumb
    -Wl,--start-group
        -lstdc++
        -lsupc++
        -lm
        -lc
        -lgcc
        -lnosys
    -Wl,--end-group
    -specs=nosys.specs
    -Wl,--wrap=main
    -Wl,--gc-sections
    -Wl,--wrap=printf
    -Wl,--wrap=sprintf
    -Wl,--wrap=snprintf
    -Wl,--wrap=vprintf
    -Wl,--wrap=vsprintf
    -Wl,--wrap=vsnprintf
    -Wl,--wrap=fprintf
    -Wl,--wrap=vfprintf
    -T${CMAKE_CURRENT_SOURCE_DIR}/mbed-os/targets/TARGET_STM/TARGET_STM32F1/TARGET_STM32F103xB/TOOLCHAIN_GCC_ARM/stm32f103xb_preprocessed.ld
)

project(${APP_TARGET})
